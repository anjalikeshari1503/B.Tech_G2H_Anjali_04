#include <stdio.h>

int main() {
     int m,n ;
  printf("enter no. of rows and columns\n");
  scanf("%d%d",&m,&n);
  // input fisrt Matrix A
  int arr[m][n];
  printf("enter array elements of Matrix A\n");
  for(int i=0;i<m;i++){
      for(int j=0;j<n;j++){
          scanf("%d",&arr[i][j]);
      }
  }
  // Transpose of Matrix A
  int brr[n][m];
  for(int i=0;i<m;i++){
      for(int j=0;j<n;j++){
          brr[j][i] = arr[i][j];
      }
  }
  if(arr[m][n]==brr[n][m]){
      printf("Matrix is symmetric");
      
  }
      else if (arr[m][n]= -brr[n][m]){
          printf("Matix is unsymmetric");
      }
      else {
          printf("Matrix is not symmetrix");
      }
    
  
    return 0;
}

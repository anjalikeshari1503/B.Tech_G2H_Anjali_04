#include <stdio.h>
#include<math.h>
int main() {
   int i,j,m,n;
   int key,count=0;
  printf("enter no. of rows and columns\n");
  scanf("%d%d",&m,&n);
  // input Matrix A
  int arr[m][n];
  printf("enter array elements of Matrix A\n");
  for(i=0;i<m;i++){
      for(j=0;j<n;j++){
          scanf("%d",&arr[i][j]);
          
      }
  }
  printf("enter a key=%d",key );
  scanf("%d",&key);
  for(i=0;i<m;i++){
      for(j=0;j<n;j++){
          if(key==arr[i][j]){
              count ++ ;
          }
      }
  }
  printf("frequency of a no. in a matrix=%d",count );

    return 0;
}
